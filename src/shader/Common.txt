vec3  p2_3D(vec3 st)   
{
st.x=abs(st.x-float(int(st.x)));
st.y=abs(st.y-float(int(st.y)));
st.z=abs(st.z-float(int(st.z)));
st.x=abs(2.0*(st.x-0.5));
st.y=abs(2.0*(st.y-0.5));
st.z=abs(2.0*(st.z-0.5));
return st;
}

vec2  p2(vec2 st)   
{
st.x=abs(st.x-float(int(st.x)));
st.y=abs(st.y-float(int(st.y)));   
st.x=abs(2.0*(st.x-0.5));
st.y=abs(2.0*(st.y-0.5));
return st;
}

vec2 TexMap_Sphere(vec3 N[3], vec3 N_tex)
{
vec2 uv;
float x= dot(N[0],N_tex); 
float y= dot(N[1],N_tex); 
float z=-dot(N[2],N_tex);    
uv.y=acos(z)/(pi); 
uv.x=atan(x,y)/(2.0*pi);  
return uv;
}

float smooth_step( float min, float max, float x )
{
    float t =(x - min) / (max - min);
    t = clamp(t, 0.0, 1.0);
    t = t * t * (3.0 - 2.0 * t); // smoothstep formula   
    return t;
}

float step2( float min, float max, float x )
{
    float t =(x - min) / (max - min);
    t = clamp(t, 0.0, 1.0); 
    return t;
}

LocalCoord RandomCoord(vec2 uv)
{
    LocalCoord L;
    L.P= vec3(0.0,0.0,0.0);
    //vec3 V1=vec3(iMouse.x/iResolution.x-0.5,0.0,iMouse.y/iResolution.y-0.5);
    //V2=vec3(0.0,0.0,0.1);
    vec3 V1=random3Dvec(uv);
    vec3 V2=vec3(0.0,0.5,0.1);
    L.N[2]=V2/length(V2);
    vec3 V0=cross(V1,V2); 
    L.N[0]=V0/length(V0);
    L.N[1]=cross(L.N[2],L.N[0]); 
    L.s[0]=300.0;
    L.s[1]=300.0;
    L.s[2]=300.0;
    return L;
}

float ComputeIntersection(Ray RI, Sphere SI)
{
float B,C,delta,t; 
    B= dot(RI.N,SI.P-RI.P);
    C= dot(RI.P-SI.P,RI.P-SI.P) - SI.R*SI.R;
    delta= B*B-C; 
    t=-1.0; 
    if(delta > 0.0 && B> 0.0) t=B-sqrt(delta);  //We hit sphere i
return t;
}

float ComputeSoftShadow(Ray RI, Sphere SI, float softness)
{
float B,C,R,delta,t; 
float a=softness;
vec3 Dif;
    B= dot(RI.N,SI.P-RI.P);
    C= dot(RI.P-SI.P,RI.P-SI.P) - a*a*SI.R*SI.R;
    delta= B*B-C; 
    t=-1.0; 
    if(delta > 0.0 && B> 0.0)
    {
    if(C<0.0) t=sqrt(delta)/(2.0*a*SI.R)-1.0;
    else t=1.0-sqrt(delta)/(2.0*a*SI.R);
    t=pow(t,2.0)+0.0;
    t=smoothstep(0.0,1.0,t);
    }
return t;
}

float ComputeDoverR(Ray RI, Sphere SI, float softness)
{
float B,C,R,delta,t; 
float a=softness;
vec3 Dif;
    B= dot(RI.N,SI.P-RI.P);
    C= dot(RI.P-SI.P,RI.P-SI.P) - a*a*SI.R*SI.R;
    delta= B*B-C; 
    R=0.0; 
    if(delta > 0.0 && B> 0.0)
    {
    if(C<0.0) R=-B+sqrt(delta);
    else R=B-sqrt(delta);
    }
return R;
}

float ComputeSoftShadow2(Ray RI, Sphere SI,float softness)
{
float B,C,R,delta,t;
float a=softness;
vec3 Dif;
    B= dot(RI.N,SI.P-RI.P);

    delta= B*B-C;

    R = length(Dif);
    t=max((R/SI.R-1.0),0.0); 
    //t=R/SI.R;
    //t=min(t,1.0);
    t=pow(100.0,-t);
    if(B< 0.0) t=-1.0;
return t;
}


Refraction ComputeRefractionComponents(vec3 N, vec3 V, float ior)
{
Refraction Data;
float fc[3]= float[3](0.1,0.00,2.50);
float Cos= dot(N,-V);
Data.R=V+2.0*Cos*N;
Data.T=1.0/ior*(-V+(Cos-sqrt(Cos*Cos-1.0+ior*ior))*N);
 //fake Fresnel
Data.F=fc[2]*(1.0-Cos)*(1.0-Cos)+fc[1]*2.0*Cos*(1.0-Cos)+fc[0]*Cos*Cos;
Data.F=clamp(Data.F,0.0,1.0);
return Data;
}  

vec3 compute_HS_Orientation(vec3 N, int j, vec2 uv, HemiSphere HS)
{
    vec3 N2t=N; 
    vec3 V1=random3Dvec(uv);
    vec3 V0=cross(V1,N2t); 
    vec3 N0t=V0/length(V0);
    vec3 N1t=cross(N2t,N0t); 
    vec3 Dir=HS.N[j];
    vec3 N_tex=Dir.x*N0t+Dir.y*N1t+Dir.z*N2t;
    return N_tex;
}
