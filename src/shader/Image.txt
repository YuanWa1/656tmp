​/*
This code provides examples for Image Based Lighting with Infinite Sphere & Final Gathering with Igloo Sampling
scene consists of one normal sphere and one environment sphere (infinite sphere)
iChannel0: Texture Image mapped to Environment Sphere
iChannel1: For keyboard control
iChannel2: Normal map mapped to the spheres
iChannel3: Texture mapped to the spheres for ambient and diffuse
*/

vec4 ComputeRefractionColor(vec3 N[3], Refraction Obj, float blur)
{
vec4 spec;
vec2 uv_tex=TexMap_Sphere(N,Obj.R);
spec = textureLod(iChannel0, uv_tex,blur);
uv_tex=TexMap_Sphere(N,-Obj.T);
vec4 refract = textureLod(iChannel0, uv_tex,blur);
spec=(1.0-Obj.F)*refract+Obj.F*spec;    
return spec;
}  

HemiSphere IglooSampleCreate(int noL1, vec2 uv, float amount)
{
int noL0;
int count=0;
float theta,phi; 
HemiSphere Igloo;
    for (float vi=0.1; vi<0.5; vi=vi+0.4/float(noL1)) 
    {
    noL0=int(1.0+4.0*float(noL1)*sqrt(1.0-sin(pi*vi)*sin(pi*vi)));
    for (float ui=0.0; ui<1.2; ui=ui+1.0/float(noL0))
    {    
    theta=2.0*pi*(ui+amount*random2Dvec(uv).x);
    phi=pi*(vi+amount*random2Dvec(uv).y);
    Igloo.N[count]=vec3(sin(theta)*sin(phi),cos(theta)*sin(phi),cos(phi));
    count++; 
    }}
    Igloo.M=count;
    return Igloo;
}


HemiSphere MercatorSampleCreate(int noL1, vec2 uv, float amount)
{
int noL0;
int count=0; 
float theta,phi; 
HemiSphere Mercator;
    for (float vi=0.0; vi<1.0; vi=vi+1.0/float(noL1)) 
    {
    for (float ui=0.0; ui<0.7; ui=ui+1.0/float(noL1)) 
    { 
    //ui=pow(ui+0.50,1.50);
    theta=2.0*pi*(ui+amount*random2Dvec(uv).x);
    phi=pi*(vi+amount*random2Dvec(uv).y);
    Mercator.N[count]=vec3(sin(theta)*sin(phi),cos(theta)*sin(phi),cos(phi));
    count++; 
    }}
    Mercator.M=count;
    return Mercator;
}


LocalCoord TimeBasedCoord(float speed)
{
    LocalCoord L;
    L.P= vec3(0.0,0.0,0.0);
    //vec3 V1=vec3(iMouse.x/iResolution.x-0.5,0.0,iMouse.y/iResolution.y-0.5);
    //V2=vec3(0.0,0.0,0.1);
    vec3 V1=vec3(cos(iTime*speed),0.0,sin(iTime*speed));
    vec3 V2=vec3(0.0,0.5,0.1);
    L.N[2]=V2/length(V2);
    vec3 V0=cross(V1,V2); 
    L.N[0]=V0/length(V0);
    L.N[1]=cross(L.N[2],L.N[0]); 
    L.s[0]=iMouse.x*iResolution.x/iResolution.x;
    L.s[1]=iMouse.x*iResolution.y/iResolution.x;
    L.s[2]=300.0;
    return L;
}

vec4 powcolor(vec4 col0,vec4 col1,vec4 col2)
{
    col0=col0+col1;
    col0.r=pow(col0.r,col2.r);
    col0.g=pow(col0.g,col2.g);
    col0.b=pow(col0.b,col2.b);
    return col0;
}   

Shading_Point_Properties Compute_Shading_Properties(Ray RI, Sphere SI, float t, int i)
{
Shading_Point_Properties SPP;
    SPP.P= RI.P + t * RI.N; 
    SPP.N=(SPP.P-SI.P)/SI.R;
    SPP.ambient =SI.ambient; 
    SPP.dif = SI.dif;
    SPP.highlight=SI.highlight;
    SPP.K_s=SI.Ks;
    SPP.ior=SI.ior;
    SPP.name=i;
    float texturePeriod=5.0;
    vec2 uv_tex=texturePeriod*TexMap_Sphere(SI.L.N,SPP.N);
    vec3 N_buf= 2.0*textureLod(iChannel2, uv_tex, 2.0).xyz-vec3(1.0);
    vec3 N_Sh=N_buf.x*SI.L.N[0]+N_buf.y*SI.L.N[1]+N_buf.z*SI.L.N[2];   
    float Nscale[4]=float[4](1.0,0.25,1.0,10.1);
    SPP.N= (Nscale[0]*SPP.N+Nscale[1]*N_Sh);
    SPP.N= SPP.N/length(SPP.N);
    SPP.ambient= (Nscale[2]*SPP.ambient+Nscale[3]*textureLod(iChannel3, uv_tex, 2.0));
    SPP.ambient=SPP.ambient/(Nscale[2]+Nscale[3]);
    SPP.dif=SPP.ambient;
return SPP;
}
